/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    bool isPalindrome(ListNode* head) {
        ListNode * p=head;
        int i=0;
        while(p!=0)
        {
            i++;
            p=p->next;
        }
        p=head;
        int* arr=new int[i];
        for(int j=0; j<i; j++)
        {
            arr[j]=p->val;
            p=p->next;
        }
        //bool ispalindrome=true;
        for(int k=0; k<i; k++)
        {
            if(arr[k]!=arr[i-k-1])
            {
                delete[] arr;
                return false;
            }
        }
        delete[] arr;
        return true;
    }
};
